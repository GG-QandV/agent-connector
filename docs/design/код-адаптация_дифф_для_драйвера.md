```rust
use serde::{Deserialize, Serialize};

#[derive(Clone, Copy, Debug, PartialEq, Eq, Default, Serialize, Deserialize)]
#[serde(rename_all = "lowercase")]
pub enum A2aWireFormat {
    #[default]
    Sdk,
    Spec,

#[derive(Clone, Debug, Serialize, Deserialize)]
pub struct A2aClientConfig {
    pub endpoint: String,
    pub token: Option<String>,
    #[serde(default)]
    pub wire_format: A2aWireFormat,
```

```rust
pub mod sdk;
pub mod spec;

use serde_json::Value;
use crate::error::A2aClientError;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum NormalizedState {
    Submitted,
    Working,
    InputRequired,
    AuthRequired,
    Completed,
    Failed,
    Canceled,
    Rejected,
}

#[derive(Clone, Debug)]
pub struct NormalizedPart {
    pub text: Option<String>,
    pub raw: Option<Vec<u8>>,
    pub uri: Option<String>,
    pub mime_type: Option<String>,
}

#[derive(Clone, Debug)]
pub struct NormalizedTask {
    pub id: String,
    pub context_id: Option<String>,
    pub state: NormalizedState,
    pub message: Option<String>,
    pub output_parts: Vec<NormalizedPart>,
}

pub trait A2aWire: Send + Sync {
    fn jsonrpc_method(&self) -> &'static str;
    fn build_message_params(&self, parts: &[NormalizedPart], context_id: Option<&str>, task_id: Option<&str>) -> Value;
    fn parse_response(&self, result: &Value) -> Result<NormalizedTask, A2aClientError>;
}
```

### Модуль `crates/driver-a2a-client/src/wire/sdk.rs`

Реализация протокола для официального SDK `a2a-rs` (`SendMessage`, proto-поля, обертка `task`):[^1]

```rust
use super::{A2aWire, NormalizedPart, NormalizedState, NormalizedTask};
use crate::error::A2aClientError;
use serde_json::{json, Value};

pub struct A2aSdkWire;

impl A2aWire for A2aSdkWire {
    fn jsonrpc_method(&self) -> &'static str {
        "SendMessage"
    }

    fn build_message_params(&self, parts: &[NormalizedPart], context_id: Option<&str>, task_id: Option<&str>) -> Value {
        let sdk_parts: Vec<Value> = parts
            .iter()
            .map(|p| {
                if let Some(ref text) = p.text {
                    json!({ "text": text })
                } else if let Some(ref uri) = p.uri {
                    json!({ "url": uri, "media_type": p.mime_type })
                } else {
                    json!({ "text": "" })
                }
            })
            .collect();

        let mut msg_obj = json!({
            "role": "ROLE_USER",
            "parts": sdk_parts
        });

        let mut params = json!({
            "message": msg_obj
        });

        if let Some(cid) = context_id {
            params.as_object_mut().unwrap().insert("contextId".to_string(), json!(cid));
        }
        if let Some(tid) = task_id {
            params.as_object_mut().unwrap().insert("taskId".to_string(), json!(tid));
        }

        params
    }

    fn parse_response(&self, result: &Value) -> Result<NormalizedTask, A2aClientError> {
        let task_val = result.get("task").ok_or_else(|| {
            A2aClientError::ProtocolError("SDK wire expected 'result.task' wrapper".into())
        })?;

        let id = task_val.get("id")
            .and_then(Value::as_str)
            .ok_or_else(|| A2aClientError::ProtocolError("Missing task.id".into()))?
            .to_string();

        let context_id = task_val.get("contextId").or_else(|| task_val.get("context_id"))
            .and_then(Value::as_str)
            .map(ToString::to_string);

        let state_raw = task_val.pointer("/status/state")
            .and_then(Value::as_str)
            .unwrap_or("TASK_STATE_UNSPECIFIED");

        let state = match state_raw {
            "TASK_STATE_COMPLETED" => NormalizedState::Completed,
            "TASK_STATE_WORKING" => NormalizedState::Working,
            "TASK_STATE_INPUT_REQUIRED" => NormalizedState::InputRequired,
            "TASK_STATE_AUTH_REQUIRED" => NormalizedState::AuthRequired,
            "TASK_STATE_FAILED" => NormalizedState::Failed,
            "TASK_STATE_CANCELLED" | "TASK_STATE_CANCELED" => NormalizedState::Canceled,
            "TASK_STATE_REJECTED" => NormalizedState::Rejected,
            _ => NormalizedState::Submitted,
        };

        let mut output_parts = Vec::new();
        if let Some(artifacts) = task_val.get("artifacts").and_then(Value::as_array) {
            for art in artifacts {
                if let Some(parts) = art.get("parts").and_then(Value::as_array) {
                    for p in parts {
                        if let Some(text) = p.get("text").and_then(Value::as_str) {
                            output_parts.push(NormalizedPart {
                                text: Some(text.to_string()),
                                raw: None,
                                uri: None,
                                mime_type: None,
                            });
                        }
                    }
                }
            }
        }

        let message = task_val.pointer("/status/message/parts/0/text")
            .and_then(Value::as_str)
            .map(ToString::to_string);

        Ok(NormalizedTask {
            id,
            context_id,
            state,
            message,
            output_parts,
        })
    }
}
```

### Модуль `crates/driver-a2a-client/src/wire/spec.rs`

Реализация для семантического A2A-шлюза (`message/send`, плоский Task, lowercase-поля):[^2][^1]

```rust
use super::{A2aWire, NormalizedPart, NormalizedState, NormalizedTask};
use crate::error::A2aClientError;
use serde_json::{json, Value};

pub struct A2aSpecWire;

impl A2aWire for A2aSpecWire {
    fn jsonrpc_method(&self) -> &'static str {
        "message/send"
    }

    fn build_message_params(&self, parts: &[NormalizedPart], context_id: Option<&str>, _task_id: Option<&str>) -> Value {
        let spec_parts: Vec<Value> = parts
            .iter()
            .map(|p| {
                if let Some(ref text) = p.text {
                    json!({ "kind": "text", "text": text })
                } else if let Some(ref uri) = p.uri {
                    json!({ "kind": "file", "file": { "uri": uri, "mime_type": p.mime_type } })
                } else {
                    json!({ "kind": "text", "text": "" })
                }
            })
            .collect();

        let mut msg_obj = json!({
            "role": "user",
            "parts": spec_parts
        });

        let mut params = json!({
            "message": msg_obj
        });

        if let Some(cid) = context_id {
            params.as_object_mut().unwrap().insert("contextId".to_string(), json!(cid));
        }

        params
    }

    fn parse_response(&self, result: &Value) -> Result<NormalizedTask, A2aClientError> {
        let task_val = if result.get("task").is_some() {
            result.get("task").unwrap()
        } else {
            result
        };

        let id = task_val.get("id")
            .and_then(Value::as_str)
            .ok_or_else(|| A2aClientError::ProtocolError("Missing flat task id".into()))?
            .to_string();

        let context_id = task_val.get("context_id").or_else(|| task_val.get("contextId"))
            .and_then(Value::as_str)
            .map(ToString::to_string);

        let state_raw = task_val.pointer("/status/state")
            .and_then(Value::as_str)
            .unwrap_or("submitted");

        let state = match state_raw {
            "completed" => NormalizedState::Completed,
            "working" => NormalizedState::Working,
            "input_required" | "inputRequired" => NormalizedState::InputRequired,
            "auth_required" | "authRequired" => NormalizedState::AuthRequired,
            "failed" => NormalizedState::Failed,
            "canceled" | "cancelled" => NormalizedState::Canceled,
            "rejected" => NormalizedState::Rejected,
            _ => NormalizedState::Submitted,
        };

        let mut output_parts = Vec::new();
        if let Some(artifacts) = task_val.get("artifacts").and_then(Value::as_array) {
            for art in artifacts {
                if let Some(parts) = art.get("parts").and_then(Value::as_array) {
                    for p in parts {
                        if p.get("kind").and_then(Value::as_str) == Some("text") || p.get("text").is_some() {
                            if let Some(text) = p.get("text").and_then(Value::as_str) {
                                output_parts.push(NormalizedPart {
                                    text: Some(text.to_string()),
                                    raw: None,
                                    uri: None,
                                    mime_type: None,
                                });
                            }
                        }
                    }
                }
            }
        }

        let message = task_val.pointer("/status/message/parts/0/text")
            .and_then(Value::as_str)
            .map(ToString::to_string);

        Ok(NormalizedTask {
            id,
            context_id,
            state,
            message,
            output_parts,
        })
    }
}
```

### Интеграция в драйвер `crates/driver-a2a-client/src/lib.rs`

Драйвер выбирает стратегию при инициализации:[^1]

```rust
pub mod error;
pub mod wire;

use std::sync::Arc;
use adapterd_config::{A2aClientConfig, A2aWireFormat};
use error::A2aClientError;
use wire::{sdk::A2aSdkWire, spec::A2aSpecWire, A2aWire, NormalizedPart, NormalizedState, NormalizedTask};
use serde_json::{json, Value};

pub struct A2aClientDriver {
    config: A2aClientConfig,
    client: reqwest::Client,
    wire: Arc<dyn A2aWire>,
}

impl A2aClientDriver {
    pub fn new(config: A2aClientConfig) -> Self {
        let wire: Arc<dyn A2aWire> = match config.wire_format {
            A2aWireFormat::Sdk => Arc::new(A2aSdkWire),
            A2aWireFormat::Spec => Arc::new(A2aSpecWire),
        };

        Self {
            config,
            client: reqwest::Client::new(),
            wire,
        }
    }

    pub async fn send_message(
        &self,
        text: &str,
        context_id: Option<&str>,
        task_id: Option<&str>,
    ) -> Result<NormalizedTask, A2aClientError> {
        let parts = vec![NormalizedPart {
            text: Some(text.to_string()),
            raw: None,
            uri: None,
            mime_type: None,
        }];

        let params = self.wire.build_message_params(&parts, context_id, task_id);
        let payload = json!({
            "jsonrpc": "2.0",
            "id": 1,
            "method": self.wire.jsonrpc_method(),
            "params": params
        });

        let mut req = self.client.post(&self.config.endpoint).json(&payload);
        if let Some(ref token) = self.config.token {
            req = req.bearer_auth(token);
        }

        let resp = req.send().await.map_err(|e| A2aClientError::Http(e.to_string()))?;
        let body: Value = resp.json().await.map_err(|e| A2aClientError::Http(e.to_string()))?;

        if let Some(err) = body.get("error") {
            let code = err.get("code").and_then(Value::as_i64).unwrap_or(-32000);
            let msg = err.get("message").and_then(Value::as_str).unwrap_or("Unknown error");
            if code == -32601 {
                return Err(A2aClientError::MethodNotFound(format!(
                    "Method '{}' not found. Check if endpoint requires a different wire_format (current: {:?}). Server error: {}",
                    self.wire.jsonrpc_method(),
                    self.config.wire_format,
                    msg
                )));
            }
            return Err(A2aClientError::RemoteError { code, message: msg.to_string() });
        }

        let result = body.get("result").ok_or_else(|| {
            A2aClientError::ProtocolError("Missing 'result' field in JSON-RPC response".into())
        })?;

        self.wire.parse_response(result)
    }
}
```

### Специальные тесты (`crates/driver-a2a-client/tests/wire_tests.rs`)

Тесты покрывают перекрестные проверки, различия сериализации и маппинг ошибок с подсказками:[^2][^1]

```rust
use adapterd_config::{A2aClientConfig, A2aWireFormat};
use driver_a2a_client::{
    wire::{sdk::A2aSdkWire, spec::A2aSpecWire, A2aWire, NormalizedPart, NormalizedState},
    A2aClientDriver,
};
use serde_json::json;

#[test]
fn test_sdk_wire_params_and_parse() {
    let wire = A2aSdkWire;
    assert_eq!(wire.jsonrpc_method(), "SendMessage");

    let parts = vec![NormalizedPart {
        text: Some("ping".into()),
        raw: None,
        uri: None,
        mime_type: None,
    }];
    let params = wire.build_message_params(&parts, Some("ctx-1"), None);
    assert_eq!(params["message"]["role"], "ROLE_USER");
    assert_eq!(params["message"]["parts"][^0]["text"], "ping");
    assert!(params["message"]["parts"][^0].get("kind").is_none());

    let resp_payload = json!({
        "task": {
            "id": "task-100",
            "contextId": "ctx-1",
            "status": {
                "state": "TASK_STATE_COMPLETED"
            },
            "artifacts": [{
                "parts": [{ "text": "pong" }]
            }]
        }
    });

    let task = wire.parse_response(&resp_payload).expect("must parse SDK task");
    assert_eq!(task.id, "task-100");
    assert_eq!(task.state, NormalizedState::Completed);
    assert_eq!(task.output_parts[^0].text.as_deref(), Some("pong"));
}

#[test]
fn test_spec_wire_params_and_parse() {
    let wire = A2aSpecWire;
    assert_eq!(wire.jsonrpc_method(), "message/send");

    let parts = vec![NormalizedPart {
        text: Some("ping".into()),
        raw: None,
        uri: None,
        mime_type: None,
    }];
    let params = wire.build_message_params(&parts, Some("ctx-1"), None);
    assert_eq!(params["message"]["role"], "user");
    assert_eq!(params["message"]["parts"][^0]["kind"], "text");
    assert_eq!(params["message"]["parts"][^0]["text"], "ping");

    let resp_payload = json!({
        "id": "task-200",
        "context_id": "ctx-1",
        "status": {
            "state": "completed"
        },
        "artifacts": [{
            "parts": [{ "kind": "text", "text": "pong" }]
        }]
    });

    let task = wire.parse_response(&resp_payload).expect("must parse Spec task");
    assert_eq!(task.id, "task-200");
    assert_eq!(task.state, NormalizedState::Completed);
    assert_eq!(task.output_parts[^0].text.as_deref(), Some("pong"));
}

#[tokio::test]
async fn test_method_not_found_hint_propagation() {
    use wiremock::matchers::{method, path};
    use wiremock::{Mock, MockServer, ResponseTemplate};

    let server = MockServer::start().await;
    Mock::given(method("POST"))
        .and(path("/rpc"))
        .respond_with(ResponseTemplate::new(200).set_body_json(json!({
            "jsonrpc": "2.0",
            "id": 1,
            "error": {
                "code": -32601,
                "message": "Method not found"
            }
        })))
        .mount(&server)
        .await;

    let driver = A2aClientDriver::new(A2aClientConfig {
        endpoint: format!("{}/rpc", server.uri()),
        token: None,
        wire_format: A2aWireFormat::Sdk,
        timeout_secs: 5,
    });

    let res = driver.send_message("hello", None, None).await;
    assert!(res.is_err());
    let err_msg = res.unwrap_err().to_string();
    assert!(err_msg.contains("wire_format"));
    assert!(err_msg.contains("Method 'SendMessage' not found"));
}
```
